### Step 2: Build v4.1 Contexts

**Location:** Around line 258, after `speakerZoneContext` is built

**Add:**
```typescript
  // NEW v4.1: Text alignment - center headlines, left details
  const textAlignmentContext = buildTextAlignmentContext({
    headlines: 'center',
    subtitles: 'center',
    details: 'left',
    footer: 'center',
  })

  // NEW v4.1: Text shadow for white text on photos
  const textShadowContext = buildTextShadowContext({
    enabled: true,
    roles: ['headline', 'subheadline'],
    intensity: 'subtle',
  })

  // NEW v4.1: Header logo band for Yi triple-logo layout
  const headerBandContext = buildHeaderLogoBandContext({
    enabled: true,
    heightPercent: 12,
    backgroundStyle: 'white rounded rectangle container',
    logoLayout: 'three logos positioned horizontally: Yi logo on the left, Bharat Rising logo in the center, CII logo on the right',
    secondaryLogos: !!options.verticalId,  // Include vertical logos if applicable
  })

  // NEW v4.1: Footer with Yi chapter branding
  const footerStyleContext = buildFooterStyleContext(
    {
      enabled: true,
      heightPercent: 10,
      leftSection: 'standard_yi',
      rightSection: 'partner_logo',
      chapterDetails: {
        chapterName: options.organizationContext?.name || 'Kanniyakumari',
        // Hashtag and social handle auto-generated from chapter name
      },
      partnerInfo: {
        partnerLabel: 'Digital Partner',
      },
    },
    options.footerContext  // Contact info (phone, email, website, social)
  )

  // NEW v4.1: Event details card (if date/time/venue present)
  const eventDetailsCardContext = buildEventDetailsCardContext(
    {
      enabled: !!(data.eventDate || data.eventTime || data.venue),
      position: 'bottom-center',
      includeIcons: true,
      backgroundColor: 'white',
    },
    {
      date: data.eventDate,
      time: data.eventTime,
      venue: data.venue,
    }
  )
```
